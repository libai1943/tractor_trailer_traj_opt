cmake_minimum_required(VERSION 3.16)
project(tractor_trailer_planner LANGUAGES CXX)
find_package(casadi CONFIG REQUIRED)
add_library(guiding_search SHARED IMPORTED)
if(WIN32)
  set_target_properties(guiding_search PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/lib/windows-x86_64/guiding_search.dll" IMPORTED_IMPLIB "${CMAKE_CURRENT_SOURCE_DIR}/lib/windows-x86_64/libguiding_search.dll.a")
else()
  set_target_properties(guiding_search PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/lib/linux-x86_64/libguiding_search.so")
endif()
add_executable(trailer_demo src/main.cpp src/geometry.cpp src/optimizer.cpp src/results.cpp)
target_include_directories(trailer_demo PRIVATE include)
target_compile_features(trailer_demo PRIVATE cxx_std_17)
target_link_libraries(trailer_demo PRIVATE guiding_search casadi::casadi)
if(MINGW)
  target_link_options(trailer_demo PRIVATE -static-libgcc -static-libstdc++)
endif()
if(WIN32)
  add_custom_command(TARGET trailer_demo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:guiding_search> $<TARGET_FILE_DIR:trailer_demo>)
endif()
